    - alicreate format: alicreate "NAME", "PWWN"\n
    - zonecreate format: zonecreate "NAME", "MEMBER1;MEMBER2"\n
    - cfgcreate format: cfgcreate "NAME", "ZONE1;ZONE2"\n
    - Map keys sorted alphabetically before iteration
    - Unsupported members skipped; zones with zero valid members skipped with warning
    - cfgcreate member lists filtered to exclude skipped zones
    - Zone.Name field used for FOS command (not map key) to avoid @vsanN composite keys
    - cfgenable always commented: # cfgenable "NAME"  # Uncomment and run manually after verifying the config
    - Section comment headers: # --- Aliases ---, # --- Zones ---, # --- Configs ---
    - Blank line between sections for readability
    - Returns nil error on success; non-nil only on io.Writer write failures

          if scriptMode {

       func sortedMapKeys[M ~map[string]V, V any](m M) []string {
           keys := make([]string, 0, len(m))
           for k := range m {
               keys = append(keys, k)
           }
           sort.Strings(keys)
           return keys
       }
